Lab 2: Traffic Light Detection


clc;
clear;
close all;

I = imread('tf5.jpg');

subplot(2,2,1);
imshow(I);
title('Original Image');

R = I(:,:,1);
G = I(:,:,2);
B = I(:,:,3);

R = double(R);
G = double(G);
B = double(B);

redMask = (R > 150) & (R > G*1.3) & (R > B*1.3);
subplot(2,2,2);
imshow(redMask);
title('Red Signal');

yellowMask = (R > 150) & (G > 120) & (B < 100);
subplot(2,2,3);
imshow(yellowMask);
title('Yellow Signal');

greenMask = (G > 100) & (G > R*1.2) & (G > B*1.2);
subplot(2,2,4);
imshow(greenMask);
title('Green Signal');

redPixels = sum(redMask(:));
yellowPixels = sum(yellowMask(:));
greenPixels = sum(greenMask(:));

minPixels = 50;

if redPixels > minPixels && redPixels > yellowPixels && redPixels > greenPixels
    disp('RED traffic light detected');
elseif yellowPixels > minPixels && yellowPixels > redPixels && yellowPixels > greenPixels
    disp('YELLOW traffic light detected');
elseif greenPixels > minPixels && greenPixels > redPixels && greenPixels > yellowPixels
    disp('GREEN traffic light detected');
else
    disp('No traffic light detected');
end

Using imfilter

Manual Loop Implementation


clc;
clear;
close all;

img = imread('input.jpg');

if size(img, 3) == 3
    img = rgb2gray(img);
end

[rows, cols] = size(img);
mask = ones(5,5) / 25;
blurred_img = zeros(rows, cols);
p = 2;

padded_img = zeros(rows + 2*p, cols + 2*p);
padded_img(p+1:p+rows, p+1:p+cols) = double(img);

for i = 1:rows
    for j = 1:cols
        neighborhood = padded_img(i:i+4, j:j+4);
        sum_value = 0;
        for m = 1:5
            for n = 1:5
                sum_value = sum_value + neighborhood(m,n) * mask(m,n);
            end
        end
        blurred_img(i,j) = sum_value;
    end
end

blurred_img = uint8(blurred_img);

figure;
subplot(1,2,1);
imshow(img);
title('Original Image');
subplot(1,2,2);
imshow(blurred_img);
title('Blurred Image - 5x5 Averaging');

Lab 7: Mirroring an Image


clc;
clear;
close all;

i=imread('myImage.JPG');
subplot(3,2,1); imshow(i); title('originl image');

[r,c,ch]=size(i);

for ch_c = 1:ch
    for r_c = 1:r
        for c_c = 1:c
            mirrored_img(r_c, c_c, ch_c) = i(r_c, c - c_c + 1, ch_c);
        end
    end
end

subplot(3,2,2); imshow(mirrored_img); title('horizontal mirrored image');

for ch_c = 1:ch
    for r_c = 1:r
        for c_c = 1:c
            mirrored_img(r_c, c_c, ch_c) = i(r-r_c+1, c_c, ch_c);
        end
    end
end
subplot(3,2,3); imshow(mirrored_img); title('vertical mirrored image');

for ch_c = 1:ch
    for r_c = 1:r
        for c_c = 1:c
            mirrored_img(r_c, c_c, ch_c) = i(r-r_c+1, c-c_c+1, ch_c);
        end
    end
end
subplot(3,2,4); imshow(mirrored_img); title('mirrored image');
